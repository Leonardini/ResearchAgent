% Formatter: let LaTeX place floats (figures, tables, algorithms) more freely, so a float that
% does not fit leaves less empty space behind. LaTeX's defaults reserve a float page for floats
% filling at least half a page and allow floats to take at most 70% of a text page.
\renewcommand{\topfraction}{0.85}
\renewcommand{\bottomfraction}{0.7}
\renewcommand{\textfraction}{0.1}
\renewcommand{\floatpagefraction}{0.75}
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{5}
